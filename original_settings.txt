    function renderSchedulesList(schedules, locations) {
        // Group by location
        const byLocation = {};
        for (const s of schedules) {
            if (!byLocation[s.locationId]) byLocation[s.locationId] = [];
            byLocation[s.locationId].push(s);
        }

        let html = '';
        for (const [locId, scheds] of Object.entries(byLocation)) {
            const loc = locations.find(l => String(l.id) === String(locId));
            const locName = loc ? loc.name : 'Unknown Location';

            html += `<div style="font-size: 12px; font-weight: 600; color: var(--text-secondary); margin: 12px 0 6px 4px;">${Utils.escapeHTML(locName)}</div>`;

            // Sort by day, then time
            scheds.sort((a, b) => {
                if (a.dayOfWeek !== b.dayOfWeek) return a.dayOfWeek - b.dayOfWeek;
                return a.startTime.localeCompare(b.startTime);
            });

            for (const s of scheds) {
                html += `
                    <div class="schedule-item">
                        <div class="schedule-info">
                            <span class="schedule-name">${Utils.escapeHTML(s.name)} ${s.isActive === false ? '<span class="badge badge-inactive">Inactive</span>' : ''}</span>
                            <span class="schedule-detail">${Utils.getDayName(s.dayOfWeek)} Â· ${Utils.formatTime(s.startTime)} â€“ ${Utils.formatTime(s.endTime)}</span>
                        </div>
                        <div class="schedule-actions">

                            <button class="btn-icon" onclick="SettingsScreen.deleteSchedule('${s.id}')" title="Delete" style="color: var(--accent-red);">âœ•</button>
                        </div>
                    </div>
                `;
            }
        }

        return html;
    }

    // ===== PIN Handling =====
    function showPinModal(onSuccess, onCancel) {
        let pinValue = '';
        const modal = document.getElementById('pin-modal');
        const dots = document.querySelectorAll('.pin-dot');
        const errorEl = document.getElementById('pin-error');

        modal.classList.remove('hidden');
        errorEl.classList.add('hidden');
        pinValue = '';
        dots.forEach(d => d.classList.remove('filled'));

        function updateDots() {
            dots.forEach((d, i) => {
                d.classList.toggle('filled', i < pinValue.length);
            });
        }

        async function handleKey(key) {
            errorEl.classList.add('hidden');

            if (key === 'clear') {
                pinValue = '';
                updateDots();
                return;
            }

            if (key === 'back') {
                pinValue = pinValue.slice(0, -1);
                updateDots();
                return;
            }

            if (pinValue.length >= 4) return;

            pinValue += key;
            updateDots();

            if (pinValue.length === 4) {
                const storedPin = await DB.Settings.getPin();
                if (pinValue === storedPin) {
                    modal.classList.add('hidden');
                    cleanup();
                    if (onSuccess) onSuccess();
                } else {
                    errorEl.classList.remove('hidden');
                    pinValue = '';
                    setTimeout(() => updateDots(), 300);
                }
            }
        }

        // Wire up keys
        const keyHandler = (e) => {
            const key = e.target.dataset.key;
            if (key) handleKey(key);
        };

        const pad = modal.querySelector('.pin-pad');
        pad.addEventListener('click', keyHandler);

        // Keyboard support
        const kbHandler = (e) => {
            if (e.key >= '0' && e.key <= '9') handleKey(e.key);
            else if (e.key === 'Backspace') handleKey('back');
            else if (e.key === 'Escape') {
                modal.classList.add('hidden');
                cleanup();
                if (onCancel) onCancel(); else App.navigate('signin');
            }
        };
        document.addEventListener('keydown', kbHandler);

        // Cancel button
        const cancelBtn = document.getElementById('pin-cancel-btn');
        const cancelHandler = () => {
            modal.classList.add('hidden');
            cleanup();
            if (onCancel) onCancel(); else App.navigate('signin');
        };
        cancelBtn.addEventListener('click', cancelHandler);

        function cleanup() {
            pad.removeEventListener('click', keyHandler);
            document.removeEventListener('keydown', kbHandler);
            cancelBtn.removeEventListener('click', cancelHandler);
        }
    }

    function promptPin() {
        showPinModal(() => {
            isAuthenticated = true;
            render();
        });
    }

    async function checkAndPromptPin(onSuccess, onCancel) {
        const hasPin = await DB.Settings.getPin();
        if (hasPin && !isAuthenticated) {
            showPinModal(() => {
                isAuthenticated = true;
                onSuccess();
            }, onCancel);
        } else {
            onSuccess();
        }
    }

    async function changePin() {
        const newPin = prompt('Enter a new 4-digit PIN:');
        if (newPin === null) return;
        if (!/^\d{4}$/.test(newPin)) {
            alert('PIN must be exactly 4 digits.');
            return;
        }
        await DB.Settings.setPin(newPin);
        alert('PIN updated successfully.');
        await render();
    }

    async function removePin() {
        if (confirm('Remove PIN protection? Anyone will be able to access settings.')) {
            await DB.Settings.set('pin', null);
            isAuthenticated = false;
            await render();
        }
    }

    // ===== Delete Handlers =====
    async function deleteLocation(id) {
        const loc = await DB.Locations.getById(id);
        if (confirm(`Delete "${loc.name}"? Class schedules at this location will also be removed.`)) {
            // Remove related schedules
            const schedules = await DB.Schedules.getByLocation(id);
            for (const s of schedules) {
                await DB.Schedules.remove(s.id);
            }

            await DB.Locations.remove(id);

            // If this was the current location, clear it
            const currentLocId = await DB.Settings.getCurrentLocationId();
            if (currentLocId === id) {
                await DB.Settings.setCurrentLocationId(null);
                document.getElementById('header-location-name').textContent = 'No Location Set';
            }

            await render();
        }
    }

    async function deleteSchedule(id) {
        const sched = await DB.Schedules.getById(id);
        if (confirm(`Delete "${sched.name}" on ${Utils.getDayName(sched.dayOfWeek)}?`)) {
            await DB.Schedules.remove(id);
            await render();
        }
    }

    async function exportAttendanceCSV() {
        const attendance = await DB.Attendance.getAll();

    }

    async function importData(event) {
        const file = event.target.files[0];
        if (!file) return;

        if (!confirm('This will replace ALL existing data. Are you sure?')) {
            event.target.value = '';
            return;
        }

        try {
            const text = await Utils.readFileAsText(file);
            const data = JSON.parse(text);
            await DB.DataIO.importAll(data);
            alert('Data imported successfully!');
            event.target.value = '';
            await render();
        } catch (err) {
            alert('Error importing data: ' + err.message);
        }
    }

    async function deduplicateMembers() {

            return;

        try {
            const btn = document.querySelector('[onclick="SettingsScreen.deduplicateMembers()"]');
            const originalHtml = btn.innerHTML;
            btn.style.pointerEvents = 'none';
            btn.innerHTML = '<div class="settings-item-left"><span class="settings-item-title">Cleaning...</span></div>';

            const count = await DB.Members.deduplicate();
            
            btn.innerHTML = originalHtml;
            btn.style.pointerEvents = 'auto';

            if (count > 0) {
                alert(`âœ… Successfully merged ${count} duplicate records! Your member list is now clean.`);
            } else {
                alert('No duplicates found. Your member list is already clean! âœ¨');

        }
    }
